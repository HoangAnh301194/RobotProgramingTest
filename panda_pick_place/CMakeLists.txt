cmake_minimum_required(VERSION 3.8)
project(panda_pick_place)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(control_msgs REQUIRED)
find_package(gazebo_msgs REQUIRED)
find_package(linkattacher_msgs REQUIRED)

add_executable(pick_place_node src/pick_place.cpp)
ament_target_dependencies(pick_place_node
  rclcpp rclcpp_action moveit_ros_planning_interface moveit_msgs
  geometry_msgs shape_msgs control_msgs gazebo_msgs linkattacher_msgs)
target_compile_features(pick_place_node PUBLIC cxx_std_17)
install(TARGETS pick_place_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config worlds DESTINATION share/${PROJECT_NAME})
ament_package()
